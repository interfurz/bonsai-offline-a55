function(bonsai_wrap_cpu target)
    target_sources(${target} PRIVATE $<TARGET_OBJECTS:bonsai-cpu-wrappers>)
    target_link_options(${target} PRIVATE "-Wl,--wrap=ggml_backend_init_by_type")
    foreach(dot q1_0_q8_0 q2_0_q8_0 pq2_0_q8_0 pq2_0_q8_K ptq1_0_q8_0)
        target_link_options(${target} PRIVATE "-Wl,--wrap=ggml_vec_dot_${dot}")
    endforeach()
endfunction()
