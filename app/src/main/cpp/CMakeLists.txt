cmake_minimum_required(VERSION 3.22.1)
project(bonsai_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BONSAI_VULKAN "Compile the experimental Vulkan backend" OFF)
option(BONSAI_HOST_SMOKE "Speed up host-only functional test builds" OFF)
option(BONSAI_KERNEL_TESTS "Build standalone ternary CPU kernel checks" ${BONSAI_HOST_SMOKE})
option(BONSAI_MODEL_TESTS "Build standalone real-model arithmetic check" OFF)
set(BONSAI_CORE_ARCHIVES "" CACHE PATH "Reuse backend-independent core archives from the identical runtime build")
get_filename_component(BONSAI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(PRISM_DIR "${BONSAI_ROOT}/vendor/llama.cpp")
if(NOT EXISTS "${PRISM_DIR}/include/llama.h")
    message(FATAL_ERROR "PrismML source missing. Run python3 scripts/fetch_runtime.py in the project root first.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "Galaxy A55 CPU" FORCE)
set(GGML_VULKAN ${BONSAI_VULKAN} CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_NUMBER 10777 CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMIT "e3de715" CACHE STRING "" FORCE)
add_subdirectory("${PRISM_DIR}" prism)
if(NOT ANDROID AND BONSAI_HOST_SMOKE)
    # CPU kernels stay optimized; graph construction is sufficient at -O0 for this test.
    target_compile_options(llama PRIVATE -O0)
    target_compile_options(llama-common PRIVATE -O0)
endif()
add_library(bonsai-jni SHARED bonsai_jni.cpp reference_cpu.cpp)
target_link_options(bonsai-jni PRIVATE "-Wl,--wrap=ggml_backend_init_by_type")
target_include_directories(bonsai-jni PRIVATE "${PRISM_DIR}/ggml/src" "${PRISM_DIR}/ggml/src/ggml-cpu")
foreach(dot q1_0_q8_0 q2_0_q8_0 pq2_0_q8_0 pq2_0_q8_K ptq1_0_q8_0)
    target_link_options(bonsai-jni PRIVATE "-Wl,--wrap=ggml_vec_dot_${dot}")
endforeach()
if(ANDROID AND BONSAI_CORE_ARCHIVES)
    find_package(Threads REQUIRED)
    # These libraries contain no GGML_USE_VULKAN conditionals; GPU dispatch stays in the new ggml build.
    file(READ "${BONSAI_CORE_ARCHIVES}/CMakeCache.txt" BONSAI_CACHED_CONFIG)
    if(NOT BONSAI_CACHED_CONFIG MATCHES "LLAMA_BUILD_COMMIT:STRING=e3de715")
        message(FATAL_ERROR "Cached runtime archives must use PrismML commit e3de715")
    endif()
    add_library(bonsai-core STATIC IMPORTED)
    set_target_properties(bonsai-core PROPERTIES
        IMPORTED_LOCATION "${BONSAI_CORE_ARCHIVES}/prism/src/libllama.a"
        INTERFACE_INCLUDE_DIRECTORIES "${PRISM_DIR}/include"
        INTERFACE_LINK_LIBRARIES "ggml")
    add_library(bonsai-build-info STATIC IMPORTED)
    set_target_properties(bonsai-build-info PROPERTIES IMPORTED_LOCATION "${BONSAI_CORE_ARCHIVES}/prism/common/libllama-common-base.a")
    add_library(bonsai-http STATIC IMPORTED)
    set_target_properties(bonsai-http PROPERTIES IMPORTED_LOCATION "${BONSAI_CORE_ARCHIVES}/prism/vendor/cpp-httplib/libcpp-httplib.a")
    add_library(bonsai-common-cached STATIC IMPORTED)
    set_target_properties(bonsai-common-cached PROPERTIES
        IMPORTED_LOCATION "${BONSAI_CORE_ARCHIVES}/prism/common/libllama-common.a"
        INTERFACE_INCLUDE_DIRECTORIES "${PRISM_DIR}/common;${PRISM_DIR}/vendor"
        INTERFACE_LINK_LIBRARIES "bonsai-core;bonsai-build-info;bonsai-http;Threads::Threads")
    target_link_libraries(bonsai-jni PRIVATE bonsai-common-cached)
else()
    target_link_libraries(bonsai-jni PRIVATE llama llama-common)
endif()
if(ANDROID)
    target_link_libraries(bonsai-jni PRIVATE android log)
else()
    # Exercise the same JNI implementation on the build host with a real GGUF.
    find_package(JNI REQUIRED)
    target_include_directories(bonsai-jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_compile_options(bonsai-jni PRIVATE -Wall -Wextra)
target_link_options(bonsai-jni PRIVATE "-Wl,-z,max-page-size=16384")
target_compile_definitions(bonsai-jni PRIVATE BONSAI_HAS_VULKAN=$<BOOL:${BONSAI_VULKAN}>)
if(BONSAI_MODEL_TESTS)
    add_executable(bonsai-model-test "${BONSAI_ROOT}/tests/ModelArithmeticSmoke.cpp" reference_cpu.cpp)
    target_link_options(bonsai-model-test PRIVATE "-Wl,--wrap=ggml_backend_init_by_type")
    target_include_directories(bonsai-model-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${PRISM_DIR}/ggml/src" "${PRISM_DIR}/ggml/src/ggml-cpu")
    target_link_libraries(bonsai-model-test PRIVATE llama llama-common)
    foreach(dot q1_0_q8_0 q2_0_q8_0 pq2_0_q8_0 pq2_0_q8_K ptq1_0_q8_0)
        target_link_options(bonsai-model-test PRIVATE "-Wl,--wrap=ggml_vec_dot_${dot}")
    endforeach()
endif()
if(BONSAI_KERNEL_TESTS)
    add_executable(bonsai-reference-test "${BONSAI_ROOT}/tests/reference_cpu_test.cpp" reference_cpu.cpp)
    target_link_options(bonsai-reference-test PRIVATE "-Wl,--wrap=ggml_backend_init_by_type" "-Wl,--wrap=ggml_compute_forward_fwht")
    target_include_directories(bonsai-reference-test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${PRISM_DIR}/ggml/src" "${PRISM_DIR}/ggml/src/ggml-cpu")
    target_link_libraries(bonsai-reference-test PRIVATE ggml ggml-cpu ggml-base)
    foreach(dot q1_0_q8_0 q2_0_q8_0 pq2_0_q8_0 pq2_0_q8_K ptq1_0_q8_0)
        target_link_options(bonsai-reference-test PRIVATE "-Wl,--wrap=ggml_vec_dot_${dot}")
    endforeach()
endif()
